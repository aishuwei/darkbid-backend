const app = getApp()
const api = require('../../utils/api.js')

Page({
  data: {
    currentStep: 1,
    requirementType: 'text',
    requirementText: '',
    requirementFile: '',
    requirementFileName: '',
    documentFile: '',
    documentFileName: '',
    generatingRules: false,
    uploadingDoc: false,
    taskId: ''
  },

  switchType(e) {
    const type = e.currentTarget.dataset.type
    this.setData({ requirementType: type })
  },

  onTextInput(e) {
    this.setData({ requirementText: e.detail.value })
  },

  chooseReqFile() {
    wx.chooseMessageFile({
      count: 1,
      type: 'file',
      extension: ['docx', 'doc', 'txt'],
      success: (res) => {
        const file = res.tempFiles[0]
        this.setData({
          requirementFile: file.path,
          requirementFileName: file.name
        })
      }
    })
  },

  chooseReqImage() {
    wx.chooseImage({
      count: 1,
      sourceType: ['album', 'camera'],
      success: (res) => {
        this.setData({
          requirementFile: res.tempFilePaths[0],
          requirementFileName: '图片'
        })
      }
    })
  },

  generateRules() {
    const { requirementType, requirementText, requirementFile } = this.data

    if (requirementType === 'text' && !requirementText.trim()) {
      wx.showToast({ title: '请输入格式要求', icon: 'none' })
      return
    }
    if ((requirementType === 'file' || requirementType === 'image') && !requirementFile) {
      wx.showToast({ title: '请上传文件', icon: 'none' })
      return
    }

    this.setData({ generatingRules: true })
    wx.showLoading({ title: '生成配置中...' })

    // 调用后端
    if (requirementType === 'text') {
      api.uploadRequirementText(requirementText)
        .then(res => {
          if (res.success) {
            return api.generateRules(res.task_id)
          }
          throw new Error(res.message)
        })
        .then(res => {
          wx.hideLoading()
          this.setData({
            generatingRules: false,
            taskId: res.task_id,
            currentStep: 2
          })
          wx.showToast({ title: '配置已生成', icon: 'success' })
        })
        .catch(err => {
          wx.hideLoading()
          this.setData({ generatingRules: false })
          wx.showToast({ title: '错误: ' + (err.message || '请检查后端'), icon: 'none', duration: 3000 })
          console.error('生成配置失败:', err)
        })
    } else {
      // 文件上传
      api.uploadRequirementFile(requirementFile)
        .then(res => {
          if (res.success) {
            return api.generateRules(res.task_id)
          }
          throw new Error(res.message)
        })
        .then(res => {
          wx.hideLoading()
          this.setData({
            generatingRules: false,
            taskId: res.task_id,
            currentStep: 2
          })
          wx.showToast({ title: '配置已生成', icon: 'success' })
        })
        .catch(err => {
          wx.hideLoading()
          this.setData({ generatingRules: false })
          wx.showToast({ title: '错误: ' + (err.message || '请检查后端'), icon: 'none', duration: 3000 })
          console.error('生成配置失败:', err)
        })
    }
  },

  chooseDocFile() {
    wx.chooseMessageFile({
      count: 1,
      type: 'file',
      extension: ['docx'],
      success: (res) => {
        const file = res.tempFiles[0]
        if (file.size > 1048576) {
          wx.showToast({ title: '文件超过1MB限制', icon: 'none' })
          return
        }
        this.setData({
          documentFile: file.path,
          documentFileName: file.name
        })
      }
    })
  },

  startCheck() {
    const { documentFile, taskId } = this.data
    if (!documentFile) {
      wx.showToast({ title: '请上传技术文件', icon: 'none' })
      return
    }
    if (!taskId) {
      wx.showToast({ title: '请先生成配置', icon: 'none' })
      return
    }

    this.setData({ uploadingDoc: true })
    wx.showLoading({ title: '上传检测中...' })

    // 调用后端上传文件
    api.uploadDocument(documentFile, taskId)
      .then(res => {
        if (res.success) {
          // 开始检查
          return api.startCheck(taskId)
        }
        throw new Error(res.message)
      })
      .then(res => {
        wx.hideLoading()
        this.setData({ uploadingDoc: false })
        
        if (res.success) {
          app.globalData.taskId = taskId
          wx.navigateTo({
            url: '/pages/waiting/waiting?taskId=' + taskId
          })
        } else {
          wx.showToast({ title: res.message || '检查失败', icon: 'none' })
        }
      })
      .catch(err => {
        wx.hideLoading()
        this.setData({ uploadingDoc: false })
        wx.showToast({ title: '错误: ' + (err.message || '请检查后端'), icon: 'none', duration: 3000 })
        console.error('检查失败:', err)
      })
  },

  goBack() {
    this.setData({ currentStep: 1 })
  }
})
